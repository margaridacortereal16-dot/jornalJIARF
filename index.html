  const fileToBase64 = (file) => {

    return new Promise((resolve, reject) => {

      const reader = new FileReader();

      reader.onload = () => {

        const result = reader.result;

        const base64 = result.split(',')[1];

        resolve(base64);

      };

      reader.onerror = reject;

      reader.readAsDataURL(file);

  // Automatic Face Detection Engine: Native Browser API + Gemini Vision AI Fallback

  const detectFacesForPhoto = async (photo, fileObj) => {

    let detectedFaces = [];

    // 1. Try Native Browser FaceDetector API (available in Chrome/Android)

    if ('FaceDetector' in window) {

      try {

        const img = new Image();

        img.src = photo.url;

        await img.decode();

        const faceDetector = new window.FaceDetector({ maxDetectedFaces: 10, fastMode: false });

        const faces = await faceDetector.detect(img);

        if (faces && faces.length > 0) {

          detectedFaces = faces.map((f, idx) => {

            const box = f.boundingBox;

            const centerX = ((box.x + box.width / 2) / img.width) * 100;

            const centerY = ((box.y + box.height / 2) / img.height) * 100;

            const radius = Math.max(10, Math.round((Math.max(box.width, box.height) / Math.min(img.width, img.height)) * 45));

              y: Math.round(centerY),

              type: selectedMaskType,

              emoji: selectedEmoji,

              sticker: 'star',

              active: true,

              radius: Math.min(25, Math.max(10, radius))

      }

      try {

        const apiKey = "";

        const apiUrl = `https://generativelanguage.googleapis.com/v1beta/models/gemini-3-flash-preview:generateContent?key=${apiKey}`;

        const payload = {

          contents: [{

            role: "user",

For each face, find center coordinates x (0 to 100 percentage from left) and y (0 to 100 percentage from top) and estimated face size radius percentage (10 to 22%). Output strictly JSON schema: { \"faces\": [ { \"x\": number, \"y\": number, \"radius\": number } ] }" },

              {

                inlineData: {

                  mimeType: fileObj.type || "image/jpeg",

                  data: base64Data

                }

              }

            ]

          }],

          generationConfig: {

            responseMimeType: "application/json"

          }

        };

          method: 'POST',

          headers: { 'Content-Type': 'application/json' },

          body: JSON.stringify(payload)

        });

          const parsed = JSON.parse(textResult);

          if (parsed.faces && Array.isArray(parsed.faces)) {

            detectedFaces = parsed.faces.map((f, idx) => ({

              id: `f-ai-${Date.now()}-${idx}`,

              x: Math.min(95, Math.max(5, Math.round(f.x))),

              emoji: selectedEmoji,

              sticker: 'star',

              active: true,

              radius: Math.min(25, Math.max(10, Math.round(f.radius || 14)))

            }));

          }

      return {

      return;

    }

      return {

        id: `uploaded-${Date.now()}-${index}`,

        url: url,

        file: file,

        isDetecting: true,

        faces: []

      };

    });

    setPhotos(prev => [...prev, ...newPhotos]);

    if (!selectedPhotoId && newPhotos.length > 0) {

      setSelectedPhotoId(newPhotos[0].id);

      const faces = await detectFacesForPhoto(photoItem, photoItem.file);

    setPhotos(prev => prev.map(p => p.id === activePhoto.id ? { ...p, isDetecting: true } : p));

    const faces = await detectFacesForPhoto(activePhoto, activePhoto.file);

    setPhotos(prev => prev.map(p => p.id === activePhoto.id ? { ...p, faces, isDetecting: false } : p));

  // Run face detection for ALL attached photos in one click

  const runFaceDetectionForAllPhotos = async () => {

    for (const photoItem of photos) {

      setPhotos(prev => prev.map(p => p.id === photoItem.id ? { ...p, isDetecting: true } : p));

      const faces = await detectFacesForPhoto(photoItem, photoItem.file);

      setPhotos(prev => prev.map(p => p.id === photoItem.id ? { ...p, faces, isDetecting: false } : p));

    }

  };

  const handleDrop = (e) => {

    setDraggingPhoto(false);

      handleFileUpload(e.dataTransfer.files);

  const removePhoto = (id, e) => {

    const updated = photos.filter(p => p.id !== id);

    setPhotos(updated);

    if (selectedPhotoId === id) {

      setSelectedPhotoId(updated.length > 0 ? updated[0].id : null);

  // Add, select or place face mask on click over image coordinates

  const handleImageClick = (photoId, event) => {

    const clickX = event.clientX - rect.left;

    const clickY = event.clientY - rect.top;

    // Convert to relative percentage (0 - 100%)

    const pctY = Math.round((clickY / rect.height) * 100);

      // Select the clicked face mask for individual adjustments

      setSelectedFaceId(existingFace.id);

      setSelectedMaskType(existingFace.type);

      // Add new face node at exact clicked location and automatically select it

      const newFaceId = `face-${Date.now()}-${Math.random().toString(36).substr(2, 4)}`;

      const newFace = {

        id: newFaceId,

        x: pctX,

        y: pctY,

        type: selectedMaskType,

        emoji: selectedEmoji,

        active: true,

        radius: maskSize

      };

      setPhotos(prevPhotos => prevPhotos.map(p => {

        if (p.id !== photoId) return p;

        return { ...p, faces: [...p.faces, newFace] };

      }));

      setSelectedFaceId(newFaceId);

    }

  };

  // Remove a specific face mask by ID

  const removeFaceMask = (photoId, faceId, e) => {

    if (e) {

      e.stopPropagation();

      e.preventDefault();

    }

    setPhotos(prev => prev.map(p => {

      if (p.id !== photoId) return p;

  // Draw canvas for accurate preview and high-quality export

  useEffect(() => {

    const canvas = canvasRef.current;

    if (!canvas) return;

    const ctx = canvas.getContext('2d');

    if (!ctx) return;

    // Set canvas high DPI size based on ratio

    const width = 1080;

    const height = aspectRatio === '1:1' ? 1080 : 1350;

    canvas.width = width;

    canvas.height = height;

    // Clear background

    ctx.clearRect(0, 0, width, height);

    // 1. Draw Background

    if (bgStyle.type === 'solid') {

      ctx.fillStyle = bgStyle.color;

      ctx.fillRect(0, 0, width, height);

    } else if (bgStyle.type === 'gradient') {

      const grad = ctx.createLinearGradient(0, 0, width, height);

      if (bgStyle.colors && bgStyle.colors.length > 0) {

        bgStyle.colors.forEach((col, i) => {

          const stop = bgStyle.colors.length > 1 ? i / (bgStyle.colors.length - 1) : 0;

        });

        grad.addColorStop(0, '#E0F2FE');

        grad.addColorStop(1, '#FBCFE8');

      }

      ctx.fillStyle = grad;

      ctx.fillRect(0, 0, width, height);

    } else if (bgStyle.type === 'pattern') {

      ctx.fillStyle = bgStyle.color;

      ctx.fillRect(0, 0, width, height);

      ctx.fillStyle = bgStyle.patternColor;

      for (let x = 20; x < width; x += 60) {

        for (let y = 20; y < height; y += 60) {

          ctx.beginPath();

          ctx.arc(x, y, 6, 0, Math.PI * 2);

          ctx.fill();

        }

      }

    }

    if (photos.length === 0) {

      // Draw empty placeholder text on canvas

      ctx.fillStyle = '#64748B';

      ctx.font = 'bold 36px sans-serif';

      ctx.textAlign = 'center';

      ctx.fillText('Adicione até 6 fotos para criar a colagem', width / 2, height / 2);

      return;

    }

    // 2. Calculate Strict Outer Padding (Border/Margem Regra Estrita)

    const padding = Math.max(borderPadding * 2.5, 40); // Scaled for 1080px resolution

    const areaX = padding;

    const areaY = padding;

    const areaW = width - (padding * 2);

    const areaH = height - (padding * 2);

    // 3. Compute Photo Grid Rectangles

    const gridRects = computeGridRectangles(photos.length, areaX, areaY, areaW, areaH, 16);

    gridRects.forEach((rect, idx) => {

      const photo = photos[idx];

      if (!photo) return;

      const img = new Image();

      img.crossOrigin = 'anonymous';

      img.src = photo.url;

      img.onload = () => {

        ctx.save();

        // Rounded photo corners

        const rad = cornerRadius * 2;

        ctx.beginPath();

        ctx.moveTo(rect.x + rad, rect.y);

        ctx.lineTo(rect.x + rect.w - rad, rect.y);

        ctx.quadraticCurveTo(rect.x + rect.w, rect.y, rect.x + rect.w, rect.y + rad);

        ctx.lineTo(rect.x + rect.w, rect.y + rect.h - rad);

        ctx.quadraticCurveTo(rect.x + rect.w, rect.y + rect.h, rect.x + rect.w - rad, rect.y + rect.h);

        ctx.lineTo(rect.x + rad, rect.y + rect.h);

        ctx.quadraticCurveTo(rect.x, rect.y + rect.h, rect.x, rect.y + rect.h - rad);

        ctx.lineTo(rect.x, rect.y + rad);

        ctx.quadraticCurveTo(rect.x, rect.y, rect.x + rad, rect.y);

        ctx.closePath();

        ctx.clip();

        // Draw Cover Fitted Image

        const imgRatio = img.width / img.height;

        const rectRatio = rect.w / rect.h;

        let renderW, renderH, renderX, renderY;

        if (imgRatio > rectRatio) {

          renderH = rect.h;

          renderW = rect.h * imgRatio;

          renderX = rect.x - (renderW - rect.w) / 2;

          renderY = rect.y;

        } else {

          renderW = rect.w;

  function computeGridRectangles(count, x, y, w, h, gap) {

    if (count === 1) {

      return [{ x, y, w, h }];

    }

    if (count === 2) {

      // Side by side

      const halfW = (w - gap) / 2;

      return [

        { x, y, w: halfW, h },

        { x: x + halfW + gap, y, w: halfW, h }

      ];

    }

    if (count === 3) {

      // 1 Top large, 2 bottom side-by-side

      const halfH = (h - gap) / 2;

      const halfW = (w - gap) / 2;

      return [

        { x, y, w, h: halfH },

        { x, y: y + halfH + gap, w: halfW, h: halfH },

        { x: x + halfW + gap, y: y + halfH + gap, w: halfW, h: halfH }

      ];

    }

    if (count === 4) {

      // 2x2 Grid

      const halfW = (w - gap) / 2;

      const halfH = (h - gap) / 2;

      return [

        { x, y, w: halfW, h: halfH },

        { x: x + halfW + gap, y, w: halfW, h: halfH },

        { x, y: y + halfH + gap, w: halfW, h: halfH },

        { x: x + halfW + gap, y: y + halfH + gap, w: halfW, h: halfH }

      ];

    }

    if (count === 5) {

      // Top 2, Bottom 3

      const halfH = (h - gap) / 2;

      const w2 = (w - gap) / 2;

      const w3 = (w - gap * 2) / 3;

      return [

        { x, y, w: w2, h: halfH },

        { x: x + w2 + gap, y, w: w2, h: halfH },

        { x, y: y + halfH + gap, w: w3, h: halfH },

        { x: x + w3 + gap, y: y + halfH + gap, w: w3, h: halfH },

        { x: x + (w3 + gap) * 2, y: y + halfH + gap, w: w3, h: halfH }

      ];

    }

    // 6 Photos: 2x3 Grid

    const colW = (w - gap * 2) / 3;

    const rowH = (h - gap) / 2;

    const rects = [];

    for (let r = 0; r < 2; r++) {

      for (let c = 0; c < 3; c++) {

        rects.push({

          x: x + c * (colW + gap),

          y: y + r * (rowH + gap),

          w: colW,

          h: rowH

        });

      }

    }

    return rects;

  }
